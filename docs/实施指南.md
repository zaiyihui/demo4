## 2. 核心服务使用指南

### 2.1 配置备份服务

#### 基本用法

```csharp
// 注入服务
private readonly IBackupService _backupService;

public async Task BackupAsync()
{
    // 创建完整备份
    var result = await _backupService.CreateBackupAsync(BackupType.Full);

    if (result.Success)
    {
        Console.WriteLine($"备份成功: {result.BackupId}");
    }
}

// 创建差异备份
var diffResult = await _backupService.CreateBackupAsync(BackupType.Differential);

// 恢复备份
var restored = await _backupService.RestoreBackupAsync("backup_20230619_120000");

// 验证备份完整性
var isValid = await _backupService.VerifyBackupIntegrityAsync("backup_20230619_120000");

// 获取所有备份
var backups = await _backupService.GetBackupsAsync();
foreach (var backup in backups)
{
    Console.WriteLine($"{backup.Id}: {backup.CreatedAt} - {backup.SizeBytes / 1024}KB");
}
```

#### 配置自动备份

```csharp
// 在 Settings.Performance 中配置
settings.Performance.AutoBackupEnabled = true;
settings.Performance.AutoBackupIntervalHours = 24;  // 每24小时
settings.Performance.DifferentialBackupEnabled = true;  // 使用差异备份
```

### 2.2 日志服务

#### 基本用法

```csharp
// 注入服务
private readonly ILogService _logService;

// 记录不同级别的日志
_logService.Trace("详细跟踪信息");
_logService.Debug("调试信息");
_logService.Info("普通信息");
_logService.Warning("警告信息");

_logService.Info("IP: 192.168.1.100");          // 输出: IP: ***.***.***.***
```

#### 自定义日志Sink

```csharp
// 添加自定义日志输出
_logService.AddSink(new CustomLogSink());

// 移除日志输出
_logService.RemoveSink(customSink);
```

### 2.3 性能监控服务

#### 基本用法

```csharp
// 注入服务
private readonly IPerformanceMonitorService _monitor;

// 获取当前指标
var metrics = _monitor.CurrentMetrics;
Console.WriteLine($"CPU: {metrics.CpuUsagePercent:F1}%");
Console.WriteLine($"内存: {metrics.MemoryUsagePercent:F1}%");
Console.WriteLine($"FPS: {metrics.Fps:F0}");

// 记录自定义指标
_monitor.RecordMetric("CustomMetric", 42.5, MetricType.Gauge);

// 操作计时
using (_monitor.BeginTiming("DatabaseQuery"))
{
    // 执行数据库查询
    await QueryDatabaseAsync();
}

// 添加告警规则
_monitor.AddAlertRule(new AlertRule
{
    Name = "HighCPU",
    MetricName = "CpuUsagePercent",
    Operator = ComparisonOperator.GreaterThan,
    Threshold = 90,
    Severity = AlertSeverity.Warning,
    Duration = TimeSpan.FromSeconds(30)
});

// 订阅指标更新
_monitor.MetricsUpdated += (s, e) =>
{
    Console.WriteLine($"更新: CPU={e.Metrics.CpuUsagePercent:F1}%");
};

// 订阅告警
_monitor.AlertTriggered += (s, e) =>
{
    Console.WriteLine($"告警: {e.Rule.Name} = {e.CurrentValue}");
};
```

### 2.5 AI服务

#### 基本用法

```csharp
// 注入服务
private readonly IAIService _aiService;

// 获取智能建议
var suggestions = await _aiService.GetSuggestionsAsync();
foreach (var suggestion in suggestions)
{
    Console.WriteLine($"[{suggestion.Category}] {suggestion.Title}");
    Console.WriteLine($"  {suggestion.Description}");
    Console.WriteLine($"  置信度: {suggestion.Confidence:P0}");
}

// 分析硬件健康状态
var analysis = await _aiService.AnalyzeHardwareAsync();
Console.WriteLine($"整体健康: {analysis.OverallHealth}");

foreach (var component in analysis.Components)
{
    Console.WriteLine($"  {component.ComponentName}: {component.Status} ({component.HealthScore:F0}%)");
    foreach (var issue in component.Issues)
    {
        Console.WriteLine($"    - {issue}");
    }
}

// 预测异常
var predictions = await _aiService.PredictAnomaliesAsync();
foreach (var prediction in predictions)
{
    Console.WriteLine($"[{prediction.Type}] 预计 {prediction.PredictedTime:g}");
    Console.WriteLine($"  {prediction.Description}");
    Console.WriteLine($"  概率: {prediction.Probability:P0}");
}

// 处理自然语言命令
var response = await _aiService.ProcessCommandAsync("当前温度是多少？");
Console.WriteLine(response);

// 设置隐私级别
_aiService.CurrentPrivacyLevel = PrivacyLevel.High;
```

### 2.6 插件服务

#### 基本用法

```csharp
// 注入服务
private readonly IPluginService _pluginService;

// 加载插件
var plugin = await _pluginService.LoadPluginAsync("path/to/plugin.dll");
if (plugin != null)
{
    Console.WriteLine($"已加载: {plugin.Name}");
}

// 获取所有已加载的插件
var plugins = _pluginService.LoadedPlugins;
foreach (var info in plugins)
{
    Console.WriteLine($"{info.Name} v{info.Version} - {info.Status}");
}

// 启用插件
await _pluginService.EnablePluginAsync("plugin-id");

// 禁用插件
await _pluginService.DisablePluginAsync("plugin-id");

// 卸载插件
await _pluginService.UnloadPluginAsync("plugin-id");

// 检查更新
var updateInfo = await _pluginService.CheckPluginUpdateAsync("plugin-id");
if (updateInfo?.IsUpdateAvailable == true)
{
    await _pluginService.UpdatePluginAsync("plugin-id", updateInfo.LatestVersion);
}

// 订阅事件
_pluginService.PluginLoaded += (s, e) => Console.WriteLine($"加载: {e.Message}");
_pluginService.PluginEnabled += (s, e) => Console.WriteLine($"启用: {e.Message}");
_pluginService.PluginDisabled += (s, e) => Console.WriteLine($"禁用: {e.Message}");
```

## 3. 集成指南

### 3.1 在 App.axaml.cs 中注册服务

```csharp
// App.axaml.cs
private void ConfigureServices()
{
    var services = new ServiceCollection();

    // 现有服务
    services.AddSingleton<ISettingsService, SettingsService>();
    services.AddSingleton<IHardwareMonitorService, HardwareMonitorService>();

    // 新增核心服务
    services.AddSingleton<ILogService, LogService>();
    services.AddSingleton<IBackupService, BackupService>();
    services.AddSingleton<IPerformanceMonitorService, PerformanceMonitorService>();
    services.AddSingleton<ILocalizationService, LocalizationService>();
    services.AddSingleton<ICloudSyncService, CloudSyncService>();
    services.AddSingleton<IAIService, AIService>();
    services.AddSingleton<IPluginService, PluginService>();

    // 插件服务依赖AI和性能监控服务
    services.AddSingleton<IPerformanceMonitorService>(sp =>
        new PerformanceMonitorService());
    services.AddSingleton<IAIService>(sp =>
        new AIService(
            sp.GetRequiredService<IPerformanceMonitorService>(),
            sp.GetRequiredService<IHardwareMonitorService>()));

    _serviceProvider = services.BuildServiceProvider();
}
```

## 6. 常见问题

### Q: 如何添加新的翻译字符串？

A: 在 `LocalizationService.cs` 的 `InitializeTranslations()` 方法中添加：

```csharp
_translations["zh-CN"]["New.Key"] = "新翻译";
_translations["en-US"]["New.Key"] = "New Translation";
```

### Q: 如何添加自定义告警规则？

A: 调用 `AddAlertRule` 方法：

```csharp
_monitor.AddAlertRule(new AlertRule
{
    Name = "CustomAlert",
    MetricName = "CustomMetric",
    Operator = ComparisonOperator.GreaterThan,
    Threshold = 100,
    Severity = AlertSeverity.Warning
});
```

### Q: 如何创建自定义日志Sink？

A: 实现 `ILogSink` 接口：

```csharp
public class CustomLogSink : ILogSink
{
    public string Name => "Custom";
    public LogLevel MinLevel { get; set; } = LogLevel.Information;

    public Task WriteAsync(LogEntry entry)
    {
        // 自定义处理逻辑
        return Task.CompletedTask;
    }

    public void Flush() { }
}
```
